\newcommand{\inputchapterwithsupplement}[2]{%
    \begingroup
    \let\mfOriginalSection\section
    \RenewDocumentCommand{\section}{s m}{%
        \IfBooleanTF{##1}{%
            \ifstrequal{##2}{Conclusion}{%
                \input{#2}%
                % Conserver l'interligne 1,5 du texte tout en évitant qu'une
                % conclusion courte déborde seule sur une page supplémentaire.
                \titlespacing*{\section}{0pt}{0.35cm}{0.22cm}%
                \setlength{\parskip}{0pt}%
                % Les chapitres I et IV n'ont besoin que d'un ajustement local
                % de pagination ; le contenu et l'interligne du texte restent inchangés.
                \ifnum\value{chapter}=1\relax\enlargethispage{3\baselineskip}\fi%
                \ifnum\value{chapter}=4\relax\enlargethispage{\baselineskip}\fi%
            }{}%
            \mfOriginalSection*{##2}%
        }{%
            % Éviter qu'un grand titre de section reste seul au bas d'une page.
            \ifstrequal{##2}{Positionnement de la Migration Factory et choix d'intégration}{\Needspace{0.22\textheight}}{}%
            \mfOriginalSection{##2}%
        }%
    }%
    \input{#1}%
    \endgroup
}
